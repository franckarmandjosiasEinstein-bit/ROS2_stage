% ------------------------------------------------------- bar-chart helper
% \timebar{width}{driving%}{blocked%}{working%}{label}
% A single stacked bar for the time-budget figure. Percentages, not seconds:
% the point of the figure is proportion, and the seconds are in the table.
\newcommand{\timebar}[5]{%
  \begin{scope}[yshift=#1]
    \fill[okFill,draw=okGreen]        (0,0) rectangle (#2/100*60mm, 4mm);
    \fill[badFill,draw=badRed]        (#2/100*60mm,0)
        rectangle ({(#2+#3)/100*60mm}, 4mm);
    \fill[warnFill,draw=warnAmber]    ({(#2+#3)/100*60mm},0)
        rectangle ({(#2+#3+#4)/100*60mm}, 4mm);
    \node[anchor=east, font=\sffamily\scriptsize] at (-1mm, 2mm) {#5};
  \end{scope}}
